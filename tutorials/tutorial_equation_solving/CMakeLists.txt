cmake_minimum_required(VERSION 3.15)
project(AriadneEquationSolvingTutorial LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}")
find_package(AriadneEquationSolving REQUIRED)
add_executable(tutorial_equation_solving tutorial_equation_solving.cpp)
target_link_libraries(tutorial_equation_solving PRIVATE AriadneEquationSolving::ariadne-equation-solving)
